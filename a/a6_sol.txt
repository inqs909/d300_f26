## 1
WITH pc AS (
	SELECT ProductID, ProductName, CategoryID, UnitPrice,
		ROUND(AVG(UnitPrice) OVER (PARTITION BY CategoryID), 2) AS cat_avg_price
	FROM Products   
)
SELECT pc.CategoryID, pc.ProductID, pc.ProductName, pc.cat_avg_price, c.CategoryName,
	ROUND(pc.UnitPrice - pc.cat_avg_price, 2) AS price_diff
FROM pc
INNER JOIN Categories as c ON pc.CategoryID = c.CategoryID
Order by pc.CategoryID, pc.ProductID;

## 2

With ranking as (
	SELECT ProductID, ProductName, CategoryID, UnitPrice,
		row_number() OVER (PARTITION BY CategoryID ORDER BY UnitPrice DESC, ProductID ASC) AS price_row_num,
		rank() OVER (PARTITION BY CategoryID ORDER BY UnitPrice DESC) AS price_rank,
		dense_rank() OVER (PARTITION BY CategoryID ORDER BY UnitPrice DESC) AS price_dense_rank
	FROM Products   
)
SELECT r.ProductID, r.ProductName, r.CategoryID, r.UnitPrice, r.price_row_num, r.price_rank, r.price_dense_rank
from ranking as r
where r.price_dense_rank < 3;

## 3

with frig as (
select CustomerID, OrderID, OrderDate, Freight,
	lag(Freight) over (PARTITION by CustomerID ORDER BY OrderDate, OrderID) as prev_freight
from orders
where Freight is not null
)
select f.CustomerID, f.OrderDate, f.OrderID, f.Freight, f.prev_freight,
	round(f.Freight - f.prev_freight, 2) as Freight_Change
from frig as f
ORDER BY f.CustomerID, f.OrderDate, f.OrderID


## 4

with days as (
	select CustomerID, OrderID, OrderDate,
		lead(OrderDate) over (PARTITION by CustomerID ORDER BY OrderDate, OrderID) as next_order_day
	from orders
)
select CustomerID OrderID, OrderDate, next_order_day,
	round(julianday(next_order_day) - julianday(OrderDate)) as days_until_next
FROM days;

## 5

WITH daily AS (
    SELECT DATE(o.OrderDate) AS order_day,
           SUM(d.UnitPrice * d.Quantity * (1.0 - d.Discount))
               AS daily_revenue
    FROM Orders AS o
    JOIN "Order Details" AS d ON o.OrderID = d.OrderID
    WHERE o.OrderDate IS NOT NULL
    GROUP BY DATE(o.OrderDate)
)
SELECT order_day, daily_revenue,
	SUM(daily_revenue) OVER (
		ORDER BY order_day
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
	) as running_revenue,
	AVG(daily_revenue) OVER (
        ORDER BY order_day
        ROWS BETWEEN 3 PRECEDING AND CURRENT ROW
       ) as moving_avg_3
FROM daily
