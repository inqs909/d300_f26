## 1
with aa as (
	select *
	from film
	left join film_category on film.film_id = film_category.film_id
),
bb as (
	select *
	from aa
	left join category on aa.category_id = category.category_id
)
SELECT  name as category_name,
	count(case when rating = 'G' then 1 else NULL end) as g_count,
	count(case when rating = 'PG' then 1 else NULL end) as pg_count,
	count(case when rating = 'PG-13' then 1 else NULL end) as pg13_count,
	count(case when rating = 'R' then 1 else NULL end) as r_count,
	count(case when rating = 'NC-17' then 1 else NULL end) as nc17_count,
	count(*) as total_films

from bb
GROUP by name
order by name;

## 2

select 
	strftime('%m', rental_date) as rental_month,
	count(case when staff_id = 1 then 1 else NULL end) as store_1_rentals,
	count(case when staff_id = 2 then 1 else NULL end) as store_2_rentals,
	count(case when staff_id = 1 then 1 else 0 end) as total_rentals
from rental
group by  rental_month
order by rental_month;

SELECT category_name, 
       'G' AS rating, g_count AS total_count
FROM p1
UNION ALL
SELECT category_name, 
       'PG' AS rating, pg_count AS total_count
FROM p1
UNION ALL
SELECT category_name, 
       'PG-13' AS rating, pg13_count AS total_count
FROM p1
UNION ALL
SELECT category_name, 
       'R' AS rating, r_count AS total_count
FROM p1
UNION ALL
SELECT category_name, 
       'NC-17' AS rating, nc17_count AS total_count
FROM p1
ORDER BY category_name;
